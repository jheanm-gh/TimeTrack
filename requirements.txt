# TimeTrack runtime dependencies - exact pins so a rebuild in six months
# produces the same application. All of these ship prebuilt wheels for
# CPython 3.12 on Windows (no compiler needed).

# PySide6-Essentials rather than the full PySide6: identical to import, but
# without the add-on modules (WebEngine, Quick3D, Charts, Multimedia, ...)
# that this application never touches. Smaller download, faster build, and
# nothing unexpected for the packager to pick up.
PySide6-Essentials==6.8.3
openpyxl==3.1.5

# Windows has no IANA time zone database. Linux and macOS ship one in
# /usr/share/zoneinfo, so ZoneInfo("Africa/Johannesburg") works there and
# raises ZoneInfoNotFoundError on Windows. This package is that database,
# as pure Python data, and zoneinfo falls back to it automatically. Without
# it the display-timezone setting silently does nothing on Windows.
tzdata==2026.4

# Pulled in by the two above; pinned here so the whole tree is reproducible.
shiboken6==6.8.3
et-xmlfile==2.0.0
